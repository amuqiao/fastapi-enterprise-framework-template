**数据访问层封装**

```python
# app/db/repository.py
import logging
from typing import List, Optional, Type, TypeVar
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy import select, update, delete
from sqlalchemy.orm import selectinload

from app.models.database import Conversation, Message, Base

logger = logging.getLogger(__name__)
ModelType = TypeVar("ModelType", bound=Base)

class Repository:
    """通用数据访问层基类。"""

    def __init__(self, model: Type[ModelType], session: AsyncSession):
        """初始化仓储。

        Args:
            model: 数据模型类
            session: 异步数据库会话
        """
        self.model = model
        self.session = session

    async def get_by_id(self, id_value: str) -> Optional[ModelType]:
        """根据ID查询单条记录。

        Args:
            id_value: 记录ID

        Returns:
            查询到的记录，不存在返回None
        """
        try:
            query = select(self.model).where(self.model.id == id_value)
            result = await self.session.execute(query)
            return result.scalar_one_or_none()
        except Exception as e:
            logger.error(f"查询{self.model.__tablename__}失败: {e}")
            raise

    async def list_by_condition(
        self,
        offset: int = 0,
        limit: int = 100,
        **conditions
    ) -> List[ModelType]:
        """根据条件查询列表。

        Args:
            offset: 偏移量
            limit: 返回数量限制
            **conditions: 查询条件

        Returns:
            查询结果列表
        """
        try:
            query = select(self.model).offset(offset).limit(limit)
            for field, value in conditions.items():
                query = query.where(getattr(self.model, field) == value)

            result = await self.session.execute(query)
            return list(result.scalars().all())
        except Exception as e:
            logger.error(f"查询{self.model.__tablename__}列表失败: {e}")
            raise

    async def create(self, **kwargs) -> ModelType:
        """创建新记录。

        Args:
            **kwargs: 字段值

        Returns:
            创建的记录
        """
        try:
            instance = self.model(**kwargs)
            self.session.add(instance)
            await self.session.commit()
            await self.session.refresh(instance)
            logger.info(f"创建{self.model.__tablename__}成功: {instance.id}")
            return instance
        except Exception as e:
            await self.session.rollback()
            logger.error(f"创建{self.model.__tablename__}失败: {e}")
            raise

    async def update(self, id_value: str, **kwargs) -> Optional[ModelType]:
        """更新记录。

        Args:
            id_value: 记录ID
            **kwargs: 更新的字段值

        Returns:
            更新后的记录，不存在返回None
        """
        try:
            await self.session.execute(
                update(self.model)
                .where(self.model.id == id_value)
                .values(**kwargs)
            )
            await self.session.commit()
            return await self.get_by_id(id_value)
        except Exception as e:
            await self.session.rollback()
            logger.error(f"更新{self.model.__tablename__}失败: {e}")
            raise

    async def delete(self, id_value: str) -> bool:
        """删除记录。

        Args:
            id_value: 记录ID

        Returns:
            是否删除成功
        """
        try:
            instance = await self.get_by_id(id_value)
            if instance:
                await self.session.delete(instance)
                await self.session.commit()
                logger.info(f"删除{self.model.__tablename__}成功: {id_value}")
                return True
            return False
        except Exception as e:
            await self.session.rollback()
            logger.error(f"删除{self.model.__tablename__}失败: {e}")
            raise

class ConversationRepository(Repository):
    """会话数据访问层。"""

    def __init__(self, session: AsyncSession):
        super().__init__(Conversation, session)

    async def get_with_messages(self, session_id: str) -> Optional[Conversation]:
        """获取会话及其消息列表。

        Args:
            session_id: 会话ID

        Returns:
            包含消息的会话对象
        """
        try:
            query = (
                select(Conversation)
                .options(selectinload(Conversation.messages))
                .where(Conversation.session_id == session_id)
            )
            result = await self.session.execute(query)
            return result.scalar_one_or_none()
        except Exception as e:
            logger.error(f"查询会话消息失败: {e}")
            raise
```

# app/cache/redis_client.py
import logging
from typing import Optional
from contextlib import asynccontextmanager

import redis.asyncio as redis
from redis.asyncio.connection import ConnectionPool

from app.core.config import settings

logger = logging.getLogger(__name__)

# 常量定义
REDIS_MAX_CONNECTIONS = 50
REDIS_TIMEOUT = 5
REDIS_COMMAND_TIMEOUT = 3

class RedisClient:
    """Redis客户端单例类。"""

    _instance: Optional["RedisClient"] = None
    _pool: Optional[ConnectionPool] = None

    def __new__(cls):
        if cls._instance is None:
            cls._instance = super().__new__(cls)
        return cls._instance

    async def init_client(self):
        """初始化Redis连接池。"""
        if self._pool is None:
            pool_config = settings.REDIS_POOL_CONFIG
            self._pool = ConnectionPool(
                host=pool_config.host,
                port=pool_config.port,
                password=pool_config.password,
                db=pool_config.db,
                max_connections=pool_config.max_connections,
                decode_responses=True,
                socket_timeout=REDIS_TIMEOUT,
                socket_connect_timeout=REDIS_TIMEOUT,
            )
            logger.info("Redis连接池初始化成功")

    @property
    async def client(self) -> redis.Redis:
        """获取Redis客户端实例。"""
        if self._pool is None:
            await self.init_client()
        return redis.Redis(connection_pool=self._pool)

    async def close(self):
        """关闭Redis连接池。"""
        if self._pool:
            await self._pool.disconnect()
            self._pool = None
            logger.info("Redis连接池已关闭")

import re
from typing import List, Dict, Any
from dataclasses import dataclass

@dataclass
class TextChunk:
    """文本块。"""
    chunk_id: str
    content: str
    metadata: Dict[str, Any]
    start_position: int
    end_position: int

class SemanticChunker:
    """语义分块器。"""

    def __init__(
        self,
        chunk_size: int = 1000,
        chunk_overlap: int = 100,
        min_chunk_size: int = 100,
    ):
        """初始化分块器。

        Args:
            chunk_size: 目标块大小（字符数）
            chunk_overlap: 块重叠大小
            min_chunk_size: 最小块大小
        """
        self.chunk_size = chunk_size
        self.chunk_overlap = chunk_overlap
        self.min_chunk_size = min_chunk_size

    def _split_by_paragraph(self, text: str) -> List[str]:
        """按段落分割文本。"""
        # 按换行符分割，过滤空段落
        paragraphs = [p.strip() for p in re.split(r'\n\s*\n', text) if p.strip()]
        return paragraphs

    def _split_long_paragraph(self, paragraph: str) -> List[str]:
        """分割过长的段落。"""
        if len(paragraph) <= self.chunk_size:
            return [paragraph]

        chunks = []
        start = 0

        while start < len(paragraph):
            end = min(start + self.chunk_size, len(paragraph))

            # 尝试在句号处分割
            if end < len(paragraph):
                last_period = paragraph.rfind('.', start, end)
                if last_period > start + self.chunk_size // 2:
                    end = last_period + 1

            chunk = paragraph[start:end].strip()
            if chunk:
                chunks.append(chunk)

            start = end - self.chunk_overlap

        return chunks

    def chunk(self, text: str, document_id: str = "default") -> List[TextChunk]:
        """将文本分割为语义块。

        Args:
            text: 输入文本
            document_id: 文档ID

        Returns:
            文本块列表
        """
        paragraphs = self._split_by_paragraph(text)
        chunks = []
        chunk_id = 0
        current_position = 0

        for paragraph in paragraphs:
            # 先按段落分割
            paragraph_chunks = self._split_long_paragraph(paragraph)

            for chunk_content in paragraph_chunks:
                if len(chunk_content) < self.min_chunk_size:
                    # 合并过小的块
                    if chunks and len(chunks[-1].content) < self.chunk_size:
                        chunks[-1].content += f"\n\n{chunk_content}"
                        chunks[-1].end_position += len(chunk_content) + 2
                        continue

                chunk = TextChunk(
                    chunk_id=f"{document_id}_{chunk_id}",
                    content=chunk_content,
                    metadata={
                        "document_id": document_id,
                        "chunk_index": chunk_id,
                    },
                    start_position=current_position,
                    end_position=current_position + len(chunk_content),
                )
                chunks.append(chunk)
                chunk_id += 1
                current_position += len(chunk_content) + 2

        logger.info(f"文本分块完成: 文档={document_id}, 块数={len(chunks)}")
        return chunks

from pydantic import BaseModel, Field, validator

    ARRAY = "array"

    """参数Schema定义。"""

    enum: Optional[List[str]] = None

    returns: Dict[str, Any] = Field(default_factory=dict)
    examples: List[Dict[str, Any]] = Field(default_factory=list)

    @validator("required_parameters")

        """验证必需参数是否在参数定义中。"""
        if v:
            params = values.get("parameters", {})

                if param_name not in params:

# app/integration/dify/unified_router.py
import logging
from typing import Dict, Any, Optional
from enum import Enum
from dataclasses import dataclass

from app.core.config import settings

logger = logging.getLogger(__name__)

class EngineType(str, Enum):
    """引擎类型枚举。"""
    LANGCHAIN = "langchain"
    DIFY = "dify"

@dataclass
class AgentInfo:
    """智能体信息。"""
    agent_id: str
    name: str
    engine_type: EngineType
    config: Dict[str, Any]
    status: str = "active"

class UnifiedAgentRouter:
    """统一智能体路由。"""

    def __init__(self):
        self.agent_registry: Dict[str, AgentInfo] = {}
        self._init_builtin_agents()

    def _init_builtin_agents(self):
        """初始化内置智能体。"""
        # 从配置文件加载内置智能体
        for agent_config in settings.BUILTIN_AGENTS:
            self.register_agent(AgentInfo(
                agent_id=agent_config["id"],
                name=agent_config["name"],
                engine_type=EngineType(agent_config["engine"]),
                config=agent_config,
            ))

    def register_agent(self, agent: AgentInfo):
        """注册智能体。

        Args:
            agent: 智能体信息
        """
        self.agent_registry[agent.agent_id] = agent
        logger.info(f"智能体注册成功: {agent.agent_id} ({agent.engine_type})")

    async def route_request(
        self,
        agent_id: str,
        request_data: Dict[str, Any]
    ) -> Dict[str, Any]:
        """路由请求到对应的执行引擎。

        Args:
            agent_id: 智能体ID
            request_data: 请求数据

        Returns:
            执行结果
        """
        if agent_id not in self.agent_registry:
            raise AgentNotFoundError(f"智能体不存在: {agent_id}")

        agent = self.agent_registry[agent_id]

        if agent.engine_type == EngineType.LANGCHAIN:
            return await self._execute_langchain(agent, request_data)
        elif agent.engine_type == EngineType.DIFY:
            return await self._execute_dify(agent, request_data)
        else:
            raise UnsupportedEngineError(f"不支持的引擎类型: {agent.engine_type}")

    async def _execute_langchain(
        self,
        agent: AgentInfo,
        request_data: Dict[str, Any]
    ) -> Dict[str, Any]:
        """执行LangChain智能体。"""
        from app.agents.langchain_executor import LangChainExecutor

        executor = LangChainExecutor(agent.config)
        return await executor.execute(request_data)

    async def _execute_dify(
        self,
        agent: AgentInfo,
        request_data: Dict[str, Any]
    ) -> Dict[str, Any]:
        """执行Dify智能体。"""
        from app.integration.dify.dify_client import DifyClient

        client = DifyClient()
        return await client.chat(agent.config["dify_id"], request_data)
